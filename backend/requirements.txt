# --- Web Framework ---
fastapi==0.115.0
# Why FastAPI? Assignment requirement. Also: async-native, auto-generates
# OpenAPI docs, Pydantic validation built in, best Python framework for APIs.

uvicorn[standard]==0.30.6
# ASGI server for FastAPI. [standard] includes uvloop (faster event loop)
# and httptools (faster HTTP parsing).

# --- Database ---
sqlalchemy[asyncio]==2.0.35
# ORM with async support. We use it for models + query building,
# not raw SQL everywhere. asyncio extra enables async session support.

asyncpg==0.29.0
# DECISION: asyncpg over psycopg2
#   FastAPI is async. psycopg2 is synchronous — it blocks the event loop.
#   asyncpg is a pure-async Postgres driver written in C, ~3x faster than
#   psycopg2 for concurrent workloads. The only downside is it's Postgres-only
#   (no SQLite fallback for tests), but we're using Postgres everywhere.

pgvector==0.3.5
# Python bindings for pgvector. Adds Vector type to SQLAlchemy.

# --- LLM Clients ---
anthropic==0.39.0
# Official Anthropic SDK. Needed for Claude cloud provider.

openai==1.51.0
# DECISION: OpenAI SDK also works with Ollama
#   Ollama exposes an OpenAI-compatible API. By using the OpenAI SDK with
#   a custom base_url pointing to Ollama, we get a unified interface for
#   both local and OpenAI cloud models. Two providers, one client library.
#   This is why many Ollama tutorials use the OpenAI SDK — it's intentional
#   API compatibility from the Ollama team.

# --- Embeddings ---
sentence-transformers==3.1.1
# Local embedding model. No API key needed, runs on CPU.
# Downloads all-MiniLM-L6-v2 (~80MB) on first use.

# --- Numerical ---
numpy>=1.26.0
# Required by sentence-transformers and pgvector.

# --- Text Processing ---
tiktoken==0.7.0
# Token counting for context window management.
# Used to estimate chunk sizes and prevent exceeding model limits.

# --- Validation & Serialization ---
pydantic==2.9.2
pydantic-settings==2.5.2
# pydantic-settings reads .env files and validates config at startup.
# Catches missing required env vars immediately instead of at first use.

# --- HTTP ---
httpx==0.27.2
# Async HTTP client for health-checking Ollama and external APIs.
# DECISION: httpx over requests
#   requests is synchronous. In an async FastAPI app, using requests
#   blocks the event loop. httpx is the async-native equivalent.

# --- Logging ---
structlog==24.4.0
# DECISION: structlog over stdlib logging
#   Structured JSON logs are searchable, parseable, and filterable.
#   "level=error model= llama3.2:3b latency_ms=2340 error=timeout" is infinitely
#   more useful than "ERROR: Something went wrong" when debugging production.
#   TRADE-OFF: Slightly more setup than `logging.basicConfig()`.

# --- Testing ---
pytest==8.3.3
pytest-asyncio==0.24.0
httpx==0.27.2
# pytest-asyncio: test async functions. httpx: AsyncClient for API tests.

# --- Ingestion (sync DB access for scripts) ---
psycopg[binary]>=3.2.0
# DECISION: psycopg over psycopg2
#   psycopg2 requires libpq-dev and a C compiler to install if wheels are unavailable.
#   psycopg 3 (with the [binary] extra) provides modern pre-compiled wheels.
#   psycopg2 (compiled against your exact libpq version), but for a
#   demo the binary is simpler.

PyYAML==6.0.2
# For parsing YAML frontmatter in transcript files.

# --- Utilities ---
python-dotenv==1.0.1
# Loads .env file in development. In Docker, env vars come from docker-compose.
# Belt and suspenders — works in both contexts.

python-multipart==0.0.12
# Required by FastAPI for form data parsing (file uploads if needed later).
